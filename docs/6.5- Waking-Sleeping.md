***

For the waking/sleeping mechanism I am using an Atomic Waker, paired with per thread binary semaphores. The waker uses an atomic state integer, where each bit represents the state of a thread. This specific one does have the limitation that only 64 worker threads are supported. For my use case however this isn't a problem, if need-be, you can also use multiple wakers.

```cpp
namespace ytl
{
/// Maximum number of workers supported by the AtomicWaker bitmask.
inline constexpr size_t MAX_WAKER_WORKERS = 64;

/// Callback type: called with the worker index and user data to signal a worker's semaphore.
using WakeCallback = void (*)(size_t worker_index, void *user_data);

/*
 * Lock-free waker using a uint64_t bitmask to track sleeping workers.
 * Each bit represents a worker's sleep state.
 *
 * Supports targeted wake-ups: only sleeping workers are signaled.
 */
class alignas(64) AtomicWaker
{
  public:
    AtomicWaker() = default;

    AtomicWaker(WakeCallback wake_fn, void *user_data) :
        wake_fn_(wake_fn),
        user_data_(user_data)
    {
        assert(wake_fn && "AtomicWaker::AomticWaker: wake_fn must not be null.");
    }

    ~AtomicWaker() = default;

    AtomicWaker(const AtomicWaker &) = delete;
    AtomicWaker(AtomicWaker &&) = delete;

    AtomicWaker &
    operator=(const AtomicWaker &) = delete;

    AtomicWaker &
    operator=(AtomicWaker &&) = delete;

    void
    mark_sleeping(size_t worker_index)
    {
        assert(worker_index < MAX_WAKER_WORKERS);
        const uint64_t bit = uint64_t{1} << worker_index;
        sleeping_.fetch_or(bit, std::memory_order_release);
    }

    void
    mark_awake(size_t worker_index)
    {
        assert(worker_index < MAX_WAKER_WORKERS);
        const uint64_t bit = uint64_t{1} << worker_index;
        sleeping_.fetch_and(~bit, std::memory_order_release);
    }

    /*
     * Wake up to `count` sleeping workers.
     * Returns the number of workers actually woken.
     */
    size_t
    wake(size_t count)
    {
        if (count == 0)
        {
            return 0;
        }

        uint64_t expected = sleeping_.load(std::memory_order_acquire);

        while (expected != 0)
        {
            uint64_t desired = expected;
            size_t to_wake = 0;

            // Clear the lowest set bits up to `count` times.
            // (desired & (desired - 1)) clears the lowest set bit in the mask.
            while (desired != 0 && to_wake < count)
            {
                desired &= (desired - 1);
                ++to_wake;
            }

            // Attempt to clear all targeted bits in a single atomic operation.
            if (sleeping_.compare_exchange_weak(expected, desired, std::memory_order_acq_rel))
            {
                uint64_t claimed = expected ^ desired;

                while (claimed != 0)
                {
                    const int bit_index = std::countr_zero(claimed);
                    wake_fn_(static_cast<size_t>(bit_index), user_data_);
                    claimed &= (claimed - 1);
                }

                return to_wake;
            }
        }

        return 0;
    }

    void
    wake_all()
    {
        uint64_t snapshot = sleeping_.exchange(0, std::memory_order_acq_rel);

        while (snapshot != 0)
        {
            const int bit_index = std::countr_zero(snapshot);
            wake_fn_(static_cast<size_t>(bit_index), user_data_);
            snapshot &= (snapshot - 1);
        }
    }

    /*
     * Returns the amount of workers currently sleeping.
     */
    [[nodiscard]] size_t
    sleep_count() const
    {
        return static_cast<size_t>(std::popcount(sleeping_.load(std::memory_order_relaxed)));
    }

    [[nodiscard]] bool
    has_sleepers() const
    {
        return sleeping_.load(std::memory_order_acquire) != 0;
    }

  private:
    alignas(64) std::atomic<uint64_t> sleeping_{0};
    WakeCallback wake_fn_{nullptr};
    void *user_data_{nullptr};
};

} // namespace ytl
```

`wake_all()` uses a single `exchange(0, acq_rel)` rather than the CAS loop in `wake()`. When you're clearing every bit anyway, any concurrent `mark_sleeping` that races will simply find the mask already cleared and will re-mark immediately. The exchange also serves as the acquire fence to observe all preceding `mark_sleeping` stores.
